  <script>
    google.charts.load('current', {packages: ['corechart']});

    function openTab(evt, tabName) {
      var contents = document.getElementsByClassName("tab-content");
      for (let c of contents) c.classList.remove("active");
      var buttons = document.getElementsByClassName("nav-btn");
      for (let b of buttons) b.classList.remove("active");
      document.getElementById(tabName).classList.add("active");
      evt.currentTarget.classList.add("active");
      
      if(tabName === 'menu1') loadDropdownSampel();
      if(tabName === 'menu2' || tabName === 'menu3') loadDashboardAndTracking();
    }

    function loadDropdownSampel() {
      google.script.run.withSuccessHandler(function(list) {
        var select = document.getElementById("inputSampelList");
        select.innerHTML = '<option value="">-- Pilih Sampel --</option>';
        list.forEach(function(item) {
          var opt = document.createElement("option");
          opt.value = JSON.stringify(item);
          opt.text = "No: " + item.noSampel + " | Param: " + item.parameter + " (" + item.bentuk + ")";
          select.appendChild(opt);
        });
      }).getListSampel();
    }

    function submitRegistrasi() {
      var noSampel = document.getElementById("regNoSampel").value;
      
      var paramCheckboxes = document.querySelectorAll('input[name="param"]:checked');
      var parameters = Array.from(paramCheckboxes).map(cb => cb.value);

      var bentukCheckboxes = document.querySelectorAll('input[name="bentuk"]:checked');
      var bentukSampels = Array.from(bentukCheckboxes).map(cb => cb.value);

      if(!noSampel) { alert("Nomor sampel harus diisi!"); return; }
      if(parameters.length === 0) { alert("Pilih minimal satu parameter uji!"); return; }
      if(bentukSampels.length === 0) { alert("Pilih minimal satu bentuk sampel!"); return; }
      
      var data = {
        noSampel: noSampel,
        parameters: parameters,
        bentukSampels: bentukSampels
      };
      
      google.script.run.withSuccessHandler(function(res) {
        alert(res.message);
        if(res.status === "success") {
          document.getElementById("regNoSampel").value = "";
          // Uncheck checkboxes
          paramCheckboxes.forEach(cb => cb.checked = false);
          bentukCheckboxes.forEach(cb => cb.checked = false);
          loadDropdownSampel();
        }
      }).registerSampel(data);
    }

    function submitHasilUji() {
      var selectedVal = document.getElementById("inputSampelList").value;
      if(!selectedVal) { alert("Pilih sampel terlebih dahulu!"); return; }
      var sampelObj = JSON.parse(selectedVal);
      
      var data = {
        noSampel: sampelObj.noSampel,
        parameter: sampelObj.parameter,
        tglUji: document.getElementById("inTglUji").value,
        analis: document.getElementById("inAnalis").value,
        metode: document.getElementById("inMetode").value,
        nilaiHasil: document.getElementById("inNilai").value,
        satuan: document.getElementById("inSatuan").value,
        kesimpulan: document.getElementById("inKesimpulan").value,
        etilAsetat: document.getElementById("matEtil").value,
        naoh: document.getElementById("matNaoh").value,
        kh2po4: document.getElementById("matKh2po4").value,
        heksan: document.getElementById("matHeksan").value,
        hcl: document.getElementById("matHcl").value
      };
      
      google.script.run.withSuccessHandler(function(res) {
        alert(res.message);
        if(res.status === "success") {
          loadDropdownSampel();
        }
      }).saveHasilUji(data);
    }

    function loadDashboardAndTracking() {
      google.script.run.withSuccessHandler(function(data) {
        // Tanggal Terupdate
        document.getElementById("updateTimeText").innerText = "Terakhir diperbarui: " + data.updateTime;

        // Total PNBP
        document.getElementById("dashTotalPnbp").innerText = "Rp " + data.totalPnbpAll.toLocaleString();
        
        // Cards PNBP Per Parameter Kecil
        var cardsHtml = "";
        for (var param in data.pnbpPerParam) {
          cardsHtml += `<div class="small-card">
            <h5>${param}</h5>
            <p>Rp ${data.pnbpPerParam[param].toLocaleString()}</p>
          </div>`;
        }
        document.getElementById("pnbpCardsContainer").innerHTML = cardsHtml;
        
        // Render Grafik Batang Tegak Berbeda Warna
        google.charts.setOnLoadCallback(function() {
          drawBarChart(data.rekap);
        });
        
        // Laporan Total Bahan
        var bahanHtml = `
          <tr><td>Etil Asetat</td><td><b>${data.totalBahan.etil}</b> ml</td></tr>
          <tr><td>NaOH</td><td><b>${data.totalBahan.naoh}</b> ml</td></tr>
          <tr><td>KH2PO4</td><td><b>${data.totalBahan.kh2po4}</b> ml</td></tr>
          <tr><td>Heksan</td><td><b>${data.totalBahan.heksan}</b> ml</td></tr>
          <tr><td>HCl</td><td><b>${data.totalBahan.hcl}</b> ml</td></tr>
        `;
        document.getElementById("bahanTable").innerHTML = bahanHtml;
        
        // Ketelusuran Table
        var trackingHtml = "";
        data.ketelusuran.forEach(function(row) {
          var badgeColor = row.posisi === "Telah Selesai Uji" ? "#2b8a3e" : "#e67700";
          trackingHtml += `<tr>
            <td>${row.noSampel}</td>
            <td>${row.parameter}</td>
            <td><b style="color:${badgeColor}">${row.posisi}</b></td>
          </tr>`;
        });
        if(trackingHtml === "") trackingHtml = `<tr><td colspan="3">Belum ada sampel terdaftar.</td></tr>`;
        document.getElementById("ketelusuranTable").innerHTML = trackingHtml;
        
      }).getDashboardData();
    }

    function drawBarChart(rekapData) {
      var dataArray = [['Parameter', 'Jumlah Uji', { role: 'style' }]];
      var colors = ['#007bff', '#28a745', '#dc3545', '#ffc107', '#17a2b8', '#6f42c1'];
      
      var index = 0;
      for (var param in rekapData) {
        dataArray.push([param, rekapData[param], colors[index % colors.length]]);
        index++;
      }
      
      var chartData = google.visualization.arrayToDataTable(dataArray);
      var options = {
        legend: { position: 'none' },
        bar: { groupWidth: '60%' },
        vAxis: { minValue: 0, format: 'decimal' },
        chartArea: { width: '80%', height: '75%' }
      };
      
      var chart = new google.visualization.ColumnChart(document.getElementById('chartContainer'));
      chart.draw(chartData, options);
    }

    window.onload = function() {
      loadDropdownSampel();
    };
